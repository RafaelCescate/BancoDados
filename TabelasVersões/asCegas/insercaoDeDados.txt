SELECT nome, plano, contato
FROM aluno

FROM equipamento
WHERE quantidade > 0;

SELECT
    aluno.nome AS aluno,
    personal.nome AS personal,
    aluno.plano,
    aluno.contato
FROM aluno
INNER JOIN personal
    ON aluno.cpf_personal = personal.cpf_personal;

SELECT
    treino.dia,
    aluno.nome AS aluno,
    personal.nome AS personal,
    equipamento.nome AS equipamento,
    treino.serie,
    treino.repeticoes,
    treino.peso,
    treino.tipoTreinoDia
FROM treino
INNER JOIN aluno
    ON treino.cpf_aluno = aluno.cpf_aluno
INNER JOIN personal
    ON treino.cpf_personal = personal.cpf_personal
INNER JOIN equipamento
    ON treino.id_equipamento = equipamento.id_equipamento;
